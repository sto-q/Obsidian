在templates目录下面新建success.html页面

```html
<!DOCTYPE html>  
<!--引入thymeleaf命名空间，只有引入这个域名才可以使用相关的表达式-->  
<html lang="en" xmlns:th="http://www.thymeleaf.com">  
<head>  
    <meta charset="UTF-8">  
    <title>Title</title>  
</head>  
<body>  
<h1>这是一个成功页面。。。。。。。。。</h1>  
<span th:text="${username}"></span>  
<ul>  
    <li th:each="str:${list}" th:text="${str}"></li>  
</ul>  
</body>  
</html>
```

```java
@Controller  
@RequestMapping("thy")  
public class ThymeleafController {  
    @RequestMapping("hello")  
    public String hello(Model model){  
        List<String> list=new ArrayList<>();  
        list.add("a");  
        list.add("b");  
        list.add("c");  
        model.addAttribute("list",list);  
        model.addAttribute("username","eric");  
        return "success";  
    }  
}
```

```html
<!-- 语法 -->
<span th:text="${user.name}">默认显示的文字</span>

<!-- 实际效果：将 user.name 的值替换 span 的内容 -->
<span>张三</span>

<!-- 语法 -->
<tr th:each="item : ${list}">
    <td th:text="${item.name}"></td>
</tr>

<!-- item：循环变量，list：后台传来的集合 -->
<!-- 相当于 Java 的 for(Item item : list) -->


<!-- th:if：条件为 true 时显示 -->
<div th:if="${user.age >= 18}">成年人</div>

<!-- th:unless：条件为 false 时显示（取反） -->
<div th:unless="${user.age >= 18}">未成年人</div>

<!-- 三元运算符 -->
<span th:text="${user.sex == 1} ? '男' : '女'"></span>

<!-- 动态拼接 URL -->
<a th:href="@{/user/detail/{id}(id=${user.id})}">查看详情</a>

<!-- 相当于生成：/user/detail/1 -->

<!-- 带参数的 URL -->
<a th:href="@{/search(keyword=${keyword}, page=1)}">搜索</a>
<!-- 生成：/search?keyword=xxx&page=1 -->

<!-- 静态资源 -->
<img th:src="@{/images/logo.png}">

<form th:action="@{/user/save}" method="post">
    <!-- th:field 自动绑定对象属性 -->
    <input type="text" th:field="*{user.name}">
    <input type="text" th:field="*{user.age}">
</form>

<!-- 动态设置 input 的 value -->
<input type="text" th:value="${user.name}">

<!-- 动态设置 class -->
<div th:class="${status == 1} ? 'active' : 'inactive'">

<!-- 追加 class（保留原有 class） -->
<div class="base" th:classappend="${highlight} ? 'highlight'">

<!-- 抽取公共头部 -->
<div th:replace="common/header :: header"></div>

<!-- 引用 fragments 文件中的片段 -->
<div th:include="fragments/menu :: sidebar"></div>

<!-- 直接在 HTML 文本中使用 -->
<p>你好，[[${user.name}]]！</p>

<!-- 等同于 -->
<p th:text="'你好，' + ${user.name} + '！'">你好，张三！</p>
```
