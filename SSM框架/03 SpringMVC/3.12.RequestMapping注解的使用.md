@RequestMapping("demo1")

     * path/value :描述的是请求资源的url路径  
     * method :描述的是指定以什么请求方式来进行资源的访问，如果请求方式错误 会抛出405异常  
     * params :在请求当前控制器方法的时候，必须携带指定的参数如果不携带参数，报400异常  
     * @return  
     */  
    @RequestMapping(value = "hello",method = RequestMethod.GET,params = {"username","age"})  
    public String hello(){  
        return "success";  
    }  
}
