@Controller  
public class FileController {  
    @RequestMapping("download")  
    public ResponseEntity<byte[]> downloadFile(HttpSession session)throws IOException{  
        //获取servletContext对象  
        ServletContext servletContext=session.getServletContext();  
        //根据servletContext对象获取文件的真实路径  
        String realPath = servletContext.getRealPath("/img/a.jpg");  
        System.out.println(realPath);  
        //创建一个输入流  
        InputStream is=new FileInputStream(realPath);  
        //创建字节流数组对象  
        byte[] bytes=new byte[is.available()];  
        //将流读到字节流数组中  
        is.read(bytes);  
        //设置响应头信息  
        MultiValueMap<String,String> headers=new HttpHeaders();  
        headers.add("Content-Disposition","attachment;filename=car.jpg");//固定写法  
        //设置状态码  
        HttpStatus statusCode=HttpStatus.OK;  
        ResponseEntity<byte[]> responseEntity=new ResponseEntity<byte[]>(bytes,headers,statusCode);  
        //关闭输入流  
        return responseEntity;  
    }  
}

@RequestMapping("upload")  
public String fileUpload(HttpServletRequest request)throws Exception{  
    //读取文件上传的目录  
    String path = request.getSession().getServletContext().getRealPath("/uploads");  
    System.out.println("path:"+path);  
    //根据文件路径创建file对象如果不存在 自动创建  
    File file=new File(path);  
    if(!file.exists()){  
        file.mkdirs();  
    }  
    //创建磁盘文件项工厂  
    DiskFileItemFactory factory=new DiskFileItemFactory();  
    //根据磁盘文件项工厂创建文件上传对象  
    ServletFileUpload fileUpload=new ServletFileUpload(factory);  
    //根据FileUpload对象解析文件上传请求。获取fileItems对象的集合  
    List<FileItem> fileItems=fileUpload.parseRequest(request);  
    /**  
     * 表单项分为两种 ，一种是文件上传的表单项type="file  
     * 还有一种是普通表单项，比如文本表单控件 密码表单控件 单元按钮 复选框按钮。。。  
     */  
    for(FileItem fileItem:fileItems){  
        if(fileItem.isFormField()){  
            //普通表单项 不做任何操作  
        }else{  
            //获取文件名称  
            String name=fileItem.getName();  
            //执行文件上传操作  
            fileItem.write(new File(path,name));  
            //删除临时文件  
            fileItem.delete();  
        }  
    }  
    return "success";  
}

public String fileUpload(HttpServletRequest request, MultipartFile upload)throws Exception{  
    //读取文件上传的目录  
    String path = request.getSession().getServletContext().getRealPath("/uploads");  
    System.out.println("path:"+path);  
    //根据文件路径创建file对象如果不存在 自动创建  
    File file=new File(path);  
    if(!file.exists()){  
        file.mkdirs();  
    }  
    //获取文件上传的名称  
    String filename = upload.getOriginalFilename();  
    upload.transferTo(new File(path,filename));  
    return "success";  
}
