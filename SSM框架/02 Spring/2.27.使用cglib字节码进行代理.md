package com.lq.cglib;  
  
import net.sf.cglib.proxy.Enhancer;  
import net.sf.cglib.proxy.MethodInterceptor;  
import net.sf.cglib.proxy.MethodProxy;  
  
import java.lang.reflect.Method;  
  
/**  
 * 模拟代理对象购买电脑  
 */  
public class ProxyObject {  
    public static void main(String[] args) {  
        //自己购买电脑  
        Provider provider=new Provider();  
        provider.buyProduct("自己");  
  
        //模拟代理对象购买电脑  
        /**  
         * create 产生代理对象的方法  
         * 参数1 当前代理的类的字节码对象  
         * 参数2 真正干活的对象  
         */  
        Provider proxy=(Provider) Enhancer.create(Provider.class, new MethodInterceptor() {  
            /**  
             *             * @param o 当前代理对象的引用  
             * @param method 代理对象需要进行增强的对象方法  
             * @param objects 执行方法的代理对象用到的参数  
             * @param methodProxy 执行方法的代理对象 一般不用  
             * @return 和被增强的方法一样的返回值  
             * @throws Throwable  
             */            @Override  
            public Object intercept(Object o, Method method, Object[] objects, MethodProxy methodProxy) throws Throwable {  
                if(method.getName().equals("buyProduct")){  
                    method.invoke(provider,objects);  
                }  
                return null;  
            }  
        });  
        proxy.buyProduct("代理对象");  
    }  
}
