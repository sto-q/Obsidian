```applicationContext.xml fold=applicationContext.xml
<?xml version="1.0" encoding="UTF-8"?>  
<beans xmlns="http://www.springframework.org/schema/beans"  
       xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"  
       xsi:schemaLocation="http://www.springframework.org/schema/beans http://www.springframework.org/schema/beans/spring-beans.xsd">  
<!--第一种默认使用set方法进行注入，使用set方法进行注入的前提是 被管理的bean所属的类必须提供set方法-->  
<!--property标签 完成bean  
name 描述的bean的属性名称  
value 描述的bean的属性值(基本数据类型和字符串的值)  
ref 注入的是引用数据类型的值 引用的是另一个bean的值  
-->  
    <bean id="car" class="com.lq.pojo.Car">  
        <property name="brand" value="宝马"></property>  
        <property name="type" value="BWM350"></property>  
    </bean>  
    <bean id="user" class="com.lq.pojo.User">  
        <property name="username" value="lmq"></property>  
        <property name="address" value="USA"></property>  
        <property name="age" value="18"></property>  
        <property name="car" ref="car"></property>  
    </bean>  
<!-- 第二种通过构造函数的方法实现依赖注入  
constructor-arg 完成依赖注入 使用的是带参的构造函数完成的  
name 描述的bean的属性名称  
value 描述的bean的属性值(基本数据类型和字符串的值)  
ref 注入的是引用数据类型的值 引用的是另一个bean的值  
-->  
    <bean id="department" class="com.lq.pojo.Department">  
        <constructor-arg name="id" value="1"></constructor-arg>  
        <constructor-arg name="name" value="技术部"></constructor-arg>  
        <constructor-arg name="location" value="纽约"></constructor-arg>  
  
    </bean>    <bean id="emp" class="com.lq.pojo.Emp">  
        <constructor-arg name="id" value="1"></constructor-arg>  
        <constructor-arg name="name" value="lisi"></constructor-arg>  
        <constructor-arg name="age" value="18"></constructor-arg>  
        <constructor-arg name="department" ref="department"></constructor-arg>  
  
    </bean>  
  
    <!--  
 复杂类型值的注入  
-->  
    <bean id="animal" class="com.lq.pojo.Animal">  
        <!--注入数组类型的值-->  
        <property name="strs">  
        <array>        <value>str1</value>  
        <value>str2</value>  
        <value>str3</value>  
        </array>        </property>        <!--注入list-->  
        <property name="list">  
        <list>        <value>list1</value>  
        <value>list2</value>  
        <value>list3</value>  
        </list>        </property>        <!--注入set-->  
        <property name="set">  
       <set>        <value>s1</value>  
       <value>s2</value>  
        <value>s3</value>  
        </set>        </property>        <!--注入Map-->  
        <property name="map">  
        <map>        <entry key="k1" value="v1"></entry>  
        <entry key="k2" value="v2"></entry>  
        <entry key="k3" value="v3"></entry>  
        </map>        </property>        <!--注入properties-->  
        <property name="pros">  
        <props>     <prop key="p1">v1</prop>  
        <prop key="p2">v2</prop>  
        <prop key="p3">v3</prop>  
        </props>        </property>    </bean>  
</beans>
