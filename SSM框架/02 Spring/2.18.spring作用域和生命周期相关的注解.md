```java fold=生命周期的注解

@Component  
@Data  
public class User {  
    @Value("eric")  
    private String username;  
    @Value("admin")  
    private String password;  
    @Value("18")  
    private Integer age;  
  
    //bean初始化方法  
    @PostConstruct //bean 初始化需要执行的方法的注解  
    public void init(){  
        System.out.println("init初始化方法实现了");  
    }  
    //bean销毁方法  
    @PreDestroy //bean销毁的方执行法的注解  
    public void destroy(){  
        System.out.println("destroy方法实现了");  
    }  
  
}
```
