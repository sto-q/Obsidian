 ```java
@Configuration //修饰类 标识当前类是一个配置类 功能类似于applicationContext.xml  
@ComponentScan(basePackages="com.lq")//开启包扫描，指定扫描com.lq包及其子包的类 功能类似于xml中的<context:component-scan>  
public class SpringConfiguration {  
    /**  
     * Bean注解类型于bean标签 修饰一个方法，方法返回值就是我们需要管理bean类型  
     * 默认情况下 不指定bean名称 那bean名称就是方法的名称  
     * @return  
     */  
    @Bean(name = "dataSource")  
    public DataSource dataSource(){  
        ComboPooledDataSource dataSource=new ComboPooledDataSource();  
        try {  
            dataSource.setDriverClass("com.mysql.cj.jdbc.Driver");  
            dataSource.setJdbcUrl("jdbc:mysql://localhost:3306/mmm");  
            dataSource.setUser("root");  
            dataSource.setPassword("159951");  
            return dataSource;  
        } catch (PropertyVetoException e) {  
            e.printStackTrace();  
            throw new RuntimeException(e);  
        }  
    }  
  
    /**  
     * @Qualifier 可以修饰方法形参 意味可以从IOC容器里面获取指定Bean  
     * @param dataSource  
     * @return  
     */  
    @Bean(name = "runner")  
    public QueryRunner runner(@Qualifier("dataSource")DataSource dataSource){  
        QueryRunner runner = new QueryRunner(dataSource);  
        return runner;  
    }  
}
 ```
