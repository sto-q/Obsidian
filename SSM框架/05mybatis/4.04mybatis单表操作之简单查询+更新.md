```xml
    <!--  
        select 查询标签  
        id 是要写的接口的方法名称  
        resultType 描述接口的返回值类型（List set集合 那么这个属性值必须定义为接口的泛型）  
    -->  
    <select id="findAll" resultType="com.lq.pojo.User">  
        select * from user;  
    </select>  
<!--    parameterType   定义接口形参的数据类型-->  
    <select id="findUserById" resultType="com.lq.pojo.User" parameterType="int">  
        select *from user where id= #{id};  
    </select>  
    <!--  
        selectKey标签 获取自增长主键的标签  
        keyColumn：数据表中主键的字段  
        keyProperty 实体类中映射主键字段的属性  
        resultType 主键字段的数据类型  
          select last_insert_id() 是mysql的系统函数，获取新增主键的值  
    -->  
    <insert id="addUser" parameterType="com.lq.pojo.User" useGeneratedKeys="true" keyProperty="id" keyColumn="id">  
<!--    <selectKey keyColumn="id" keyProperty="id" resultType="int">-->  
<!--        select last_insert_id()-->  
<!--    </selectKey>-->  
        insert into user(username,birthday,sex,address)values (#{username},#{birthday},#{sex},#{address});  
    </insert>  
    <update id="upDateUser" parameterType="com.lq.pojo.User" >  
    update user set username = #{username},birthday = #{birthday},sex = #{sex},address = #{address} where id= #{id};  
    </update>  
  
    <delete id="delectUser" parameterType="int">  
        delete from user where id = #{id};  
    </delete>
```
