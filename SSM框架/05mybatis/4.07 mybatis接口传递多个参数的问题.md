```
  
//mybatis中存在多个参数的问题  
public List<User> findUserBySexAndAddress(String sex,String address);  
  
//使用map解决多个参数的问题  
public List<User> findUserByMap(Map<String,Object> map);
```

```xml
<!--    mybatis中存在多个参数的问题  
parameterType可以不写  
-->  
    <select id="findUserBySexAndAddress" resultType="com.lq.pojo.User">  
select * from user where sex='${param1}' and address='${param2}';  
    </select>  
  
<!--    使用map集合解决多个参数的问题  
#{}里面的值对应是map的key值  
-->  
    <select id="findUserByMap" parameterType="map" resultType="com.lq.pojo.User">  
select * from user where sex=#{sex} and address=#{address};  
    </select>

```
