```xml
<!--    批量删除用户（id）  
 当我们将数组或list集合作为参数传递给mybatis，那么mybatis会自动的将其包装在一个map集合中，用其名称作为key。value就是数组或者集合本身。  
 所有我们在获取一个数组或集合的时候 ，需要通过key值获取  
{"array",ids}  
{"list" ,list}  
-->  
    <delete id="delectUserList" parameterType="user">  
	delete from user where id in  
                 (                     
                <foreach collection="array" item="id" separator=",">  
                         #{id}  
                </foreach>  
                     )  
    </delete>
```
