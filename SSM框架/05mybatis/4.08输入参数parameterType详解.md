//根据id集合查询用户  
public List<User> findUserByIds(QueryVo queryVo);  
public List<User> findUserByIds2(List<Integer> list);

public class QueryVo {  
    public List<Integer> ids;  
    public List<Integer> getIds() {  
        return ids;  
    }  
    public void setIds(List<Integer> ids) {  
        this.ids = ids;  
    }  
}
