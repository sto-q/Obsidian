package com.lq;  
  
import com.baomidou.mybatisplus.core.conditions.query.QueryWrapper;  
import com.baomidou.mybatisplus.core.conditions.update.UpdateWrapper;  
import com.lq.mapper.UserMapper;  
import com.lq.pojo.User;  
import org.junit.jupiter.api.Test;  
import org.springframework.beans.factory.annotation.Autowired;  
import org.springframework.boot.test.context.SpringBootTest;  
  
import java.util.*;  
  
@SpringBootTest  
public class test {  
    @Autowired  
    UserMapper userMapper;  
    @Test  
    public void test1(){  
        List<User> userList = userMapper.selectList(null);  
        userList.forEach(System.out::println);  
    }  
  
    @Test  
    public void insertTest(){  
        User user=new User(null,"lmq",18,"183@qq.com");  
        System.out.println("影响的行数"+userMapper.insert(user));  
        System.out.println("id是"+user.getId());//id是2122677787191881730 主键的生成策略是mybatisplus按照雪花算法的策略生成的  
        //如果想按照自增的方法生成主键要插入@TableId  
    }  
  
    @Test  
    public void deleteTest(){  
        int a=userMapper.deleteById(2122677787191881730L);//根据id删除  
        System.out.println("删除的结果影响的行数"+a);  
    }  
    @Test  
    public void deleteList(){  
        List<Long> ids=new ArrayList<>();  
        ids.add(4L);  
        ids.add(5L);  
        int a = userMapper.deleteBatchIds(ids);  
        System.out.println("删除的结果影响的行数"+a) ;//批量删除  
    }  
  
    //根据条件进行删除(基于Map集合封装条件  
    @Test  
    public void deleteByMap(){  
        Map<String , Object> map=new HashMap<>();  
        map.put("name","lll");  
        map.put("age",100);  
        int a=userMapper.deleteByMap(map);//DELETE FROM  user WHERE name =? AND age =?  
        System.out.println("删除的结果影响的行数"+a) ;//  
    }  
  
    @Test  
    public void updateById(){//根据id更新  
        User user=new User();  
        user.setId(2L);  
        user.setAge(38);  
        user.setEmail("adnmin@qq.com");  
        int a=userMapper.updateById(user);  
        System.out.println("更新的结果影响的行数"+a) ;//UPDATE user SET age=? email=? WHERE id=?  
    }  
  
    @Test  
    public void updateByCondition(){  
        User user=new User();  
        user.setAge(11);  
        user.setEmail("LMQ@qq.com");  
        QueryWrapper wrapper=new QueryWrapper<>();  
        wrapper.eq("id",3L);  
        int a=this.userMapper.update(user,wrapper);//UPDATE user SET age =?,email =? WHERE (id=?)  
        System.out.println("更新的结果影响的行数"+a);  
    }  
    @Test  
    public void updateByCondition1(){  
        UpdateWrapper<User> wrapper= new UpdateWrapper<>();  
        wrapper.eq("id",1L).set("age",11).set("email","AAA@qq.com");  
        int a=this.userMapper.update(null,wrapper);  
        System.out.println("更新的结果影响的行数"+a);//UPDATE user SET age =?,email =? WHERE (id=?)  
    }  
  
    @Test//根据id  
    public void selectById(){  
        User user=userMapper.selectById(2L);  
        System.out.println(user);  
    }  
  
    @Test  
    public void selectByIds(){//根据主键多个查询  
        List<Long>ids=new ArrayList<>();  
        ids.add(2L);  
        ids.add(3L);  
        List<User> userList = userMapper.selectBatchIds(ids);  
        userList.forEach(System.out::println);  
    }  
  
  
    @Test//Map封装查询  
    public void selectMap(){  
        Map<String,Object>map=new HashMap<>();  
        map.put("id","as");  
        map.put("age",12);  
        List<User> list = userMapper.selectByMap(map);  
        list.forEach(System.out::println);  
    }  
  
    @Test  
    public void selectAll(){  
        List<User> userList = userMapper.selectList(null);  
        userList.forEach(System.out::println);  
    }  
  
    public void selectOne(){  
        QueryWrapper wrapper=new QueryWrapper<>();  
        wrapper.eq("name","lmq");  
        User user = userMapper.selectOne(wrapper);  
        System.out.println(user);  
    }  
}
