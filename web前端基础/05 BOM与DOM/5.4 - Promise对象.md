### 2Promise应用
```
let promise=new promise(function(resolve,reject){
//异步处理
//处理结束后调用resolve or reject
});
promise.then(function(result){
//result的值是上面调用 resolve(...)方法传入的值，可以对该结果进行相应处理
});
promise.catch(function(error){
//error的值是上面调用 reject(...)方法传入的值，可以对该结果进行相应处理
});

//链式调用
let promise=new Promise(function(resole,reject){
//异步处理
//处理结束后调用resolve or reject
}).then(function(result){
//result的值是上面调用 resolve(...)方法传入的值，可以对该结果进行相应处理
}).catch(function(error){
//error的值是上面调用 reject(...)方法传入的值，可以对该结果进行相应处理
});
```
Promose构造函数包括一个参数和一个带有resolve(解析)和reject(拒绝)两个参数的回调。在回调中执行一些操作(例如异步).如果一切正常则调用resolve，否则调用reject.
```
<body></body>
<script type="text/javascript">
function calculate(a,b){
	let promise=new Promise (function (resolve,reject){
		if(b===0){//异常处理使用reject函数进行拒绝
		reject(new Error("除数不能为0"))
		}else{//成功处理的情况使用resolve函数处理
		resolve(a/b);
		}
	});
	//这里resp接收的值是promise对象中resolve函数处理的参数值
	promise.then(function("处理成功"+resp){
	console.log(resp);
	});
	//这里error接收的值是promise对象中reject函数处理的参数值
	promise.catch(function("处理失败"+error){
	console.log(error);
	})
}
calculate(3,2);
<script>
```
## 箭头函数
箭头函数相当于java中的lambda表达式。传递的依然是实现过程
```
<body></body>
<script type="text/javascript">
function calculate(a,b){
	let promise=new Promise ( (resolve,reject)=>{
		if(b===0){//异常处理使用reject函数进行拒绝
		reject(new Error("除数不能为0"))
		}else{//成功处理的情况使用resolve函数处理
		resolve(a/b);
		}
	});
	//这里resp接收的值是promise对象中resolve函数处理的参数值
	promise.then(("处理成功"+resp)=>{
	console.log(resp);
	});
	//这里error接收的值是promise对象中reject函数处理的参数值
	promise.catch(("处理失败"+error)=>{
	console.log(error);
	})
}
calculate(3,2);
<script>
```
