 git clone <地址>  克隆仓库
git status 查看仓库状况
git add <> 提交到缓存区
git add . 提交全部内容到缓冲区
git commit 提交到本地仓库
git commit -m<提交注释>
:q!退出VIM
git push 提交到云端
git branch < name >
git branch 查看本地分支
git branch -a   查看所有分支包括远端
git checkout < name>切换分支
git checkout -b < name> 创建切换
git merge <要合并的分支>
git branch -d <分支名>   # 本地
git push origin --delete <分支名>  # 远程
git push -u origin < name> 推送分支到远端
